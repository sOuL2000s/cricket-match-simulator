                                gameState[key] = loadedState[key];
                            }
                        }
                    });

                    // Ensure player consistency on team A/B
                    ['A', 'B'].forEach(teamId => {
                        if (loadedState.teams?.[teamId]?.players && Array.isArray(loadedState.teams[teamId].players)) {
                            gameState.teams[teamId].players = loadedState.teams[teamId].players.map((loadedPlayer, idx) => {
                                const defaultRole = DEFAULT_ROLES[idx] || 'Batter';
                                const playerRole = VALID_ROLES.includes(loadedPlayer.role) ? loadedPlayer.role : defaultRole;
                                
                                const defaultPlayer = getNewPlayerStats(loadedPlayer.name || DEFAULT_PLAYERS[idx], playerRole);
                                
                                // Reset volatile current innings stats upon load
                                defaultPlayer.currentInningsR = 0; defaultPlayer.currentInningsB = 0; 
                                defaultPlayer.currentInningsFours = 0; defaultPlayer.currentInningsSixes = 0;
                                defaultPlayer.currentInningsSR = 0.0; defaultPlayer.currentInningsOut = false;

                                return {
                                    ...defaultPlayer, 
                                    ...loadedPlayer, 
                                };
                            }).slice(0, 11);
                        }
                        if (!gameState.teams[teamId].inningsScorecardHistory) {
                            gameState.teams[teamId].inningsScorecardHistory = [];
                        }
                    });

                }

                // 5. CRITICAL FIX: If League is active, ensure the current mode is set to league.
                // This ensures the navigation bar and home redirects work correctly.
                if (leagueState.isActive) {
                     gameState.currentMode = 'league';
                }

            } catch (e) {
                console.error("Error loading state from localStorage:", e);
                // Fallback to initial state if critical error during load
                gameState = getInitialGameState(); 
                careerRecords = { matchHistory: [], playerStats: {}, bestBatting: null, bestBowling: null, currentSort: { column: 'R', direction: 'desc' } };
                leagueState = getInitialLeagueState();

                // CRITICAL: Only save league state if it is explicitly active, otherwise delete old data.
                if (leagueState.isActive) {
                    // Create a mutable copy to ensure NRR is saved as a three decimal string format without polluting the live state if it's currently a number
                    const leagueStateToSave = JSON.parse(JSON.stringify(leagueState)); 

                    if (Array.isArray(leagueStateToSave.table)) {
                        leagueStateToSave.table = leagueStateToSave.table.map(t => ({
                             ...t,
                             NRR: typeof t.NRR === 'string' ? t.NRR : (t.NRR || 0.000).toFixed(3)
                        }));
                    }

            } else if (isUserBattingManually) {
                primaryActionHtml = `
                    <div id="manual-roll-controls" class="w-full text-center mb-4">
                        <h4 class="text-xl font-bold mb-3 text-gray-700">Choose Your Run Score (1-6):</h4>
                        <div class="flex justify-center space-x-3">
                            ${[1, 2, 3, 4, 5, 6].map(val => `
                                <button onclick="handleManualRoll(${val})" class="w-12 h-12 flex items-center justify-center bg-cricket-green text-white rounded-lg font-bold text-lg hover:bg-green-700 transition transform hover:scale-110">
                                    ${val}
                                </button>
                            `).join('')}
                        </div>
                        <p class="mt-4 text-sm text-gray-600">You choose the batting roll. Bowler's roll remains random.</p>
                    </div>
                `;
            } else {
                primaryActionHtml = `
                    <button id="roll-button" onclick="handleRoll()" class="btn-primary flex-grow text-2xl w-full">
                        ${isUserBatting ? 'ROLL THE DICE!' : 'Continue Computer Turn'}
                    </button>
                `;
            }
            // --- END ACTION LOGIC ---


            // --- Simulation Control Block ---
            const simulationControlHtml = gameState.matchResult ? '' : `
                <div class="w-full mt-4 flex space-x-2">
                    ${!gameState.isSimulating ? `
                        <button id="simulate-innings-button" onclick="simulateMatch('innings_end')"
                            class="btn-primary bg-yellow-600 hover:bg-yellow-700 flex-1 text-lg">
                            Simulate Current Innings
                        </button>
                        <button id="simulate-match-button" onclick="simulateMatch('match_end')"
                            class="btn-primary bg-purple-600 hover:bg-purple-700 flex-1 text-lg">
                            Simulate Match End
                        </button>
                    ` : `
                        <button id="stop-simulation-button" onclick="stopSimulation()"
                            class="btn-primary bg-cricket-red hover:bg-red-700 w-full text-2xl">
                            Stop Simulation
                        </button>
                    `}
                </div>
            `;


            return `
                <div class="max-w-5xl mx-auto">
                    <h2 class="text-4xl font-extrabold mb-6 text-center text-cricket-green">
                        Innings ${gameState.innings} ${gameState.gameMode === 'test_match' ? 'of 4' : ''} - ${battingTeam.name} Batting
                    </h2>
                    <div class="grid grid-cols-1 lg:grid-cols-3 gap-6">

                        <!-- Current Score Panel (Left/Top) -->
                        <div class="lg:col-span-1 bg-white p-6 card">
                            <h3 class="text-2xl font-bold mb-4 border-b pb-2 text-cricket-green">Live Score</h3>
                            <p class="text-5xl font-extrabold mb-2">${score.runs}/${score.wickets} ${targetDisplay}</p>
                            <p class="text-xl text-gray-600 mb-6">Overs: ${displayOvers} / ${gameState.oversPerInnings}.0</p>

                            <div class="space-y-3">
                                <p class="text-lg"><strong>On Strike:</strong> ${currentBatsmanOnStrike.name} <span class="font-mono text-cricket-red">(${currentBatsmanOnStrike.currentInningsR} R, ${currentBatsmanOnStrike.currentInningsB} B)</span></p>
                                <p class="text-lg"><strong>Non-Striker:</strong> ${currentBatsmanOffStrike.name} <span class="font-mono text-gray-500">(${currentBatsmanOffStrike.currentInningsR} R, ${currentBatsmanOffStrike.currentInningsB} B)</span></p>
                                <p class="text-lg"><strong>Bowler:</strong> ${currentBowler.name} <span class="font-mono text-gray-500">(${currentBowler.O}.${currentBowler.Balls % 6} - ${currentBowler.W}/${currentBowler.Runs})</span></p>
                            </div>
                            <div id="action-message" class="mt-4 p-3 bg-blue-100 text-blue-800 rounded-lg text-sm font-medium">
                                Waiting for the next roll...
                            </div>
                            ${gameState.gameMode === 'test_match' && gameState.testMatchState.currentLeadTrailMessage ? `
                                <div class="mt-4 p-3 bg-yellow-100 text-yellow-800 rounded-lg text-sm font-medium">
                                    ${gameState.testMatchState.currentLeadTrailMessage}
                                </div>
                            ` : ''}
                        </div>

                        <!-- Dice & Action Panel (Center) -->
                        <div class="lg:col-span-2 bg-white p-6 card flex flex-col items-center">
                            ${diceContainer}
                            <div class="w-full mt-6">
                                ${primaryActionHtml}
                            </div>
                            ${simulationControlHtml}
                        </div>

                        <!-- Full Scoreboard (Bottom) -->
                        <div class="lg:col-span-3">
                            ${renderScoreboardTable(battingTeam)}
                        </div>
                    </div>
                </div>
            `;
        };

        const renderScoreboardTable = (battingTeam) => {
            // Display players who have come to bat in the CURRENT innings, up to the next player in queue
            const players = battingTeam.players.slice(0, gameState.score.batting.nextPlayerIndex);
            const tableRows = players.map(player => `
                <tr class="text-gray-700 ${player.currentInningsOut ? 'bg-red-50' : (player.currentInningsB > 0 && !player.currentInningsOut ? 'bg-green-50' : 'bg-gray-50')}">
                    <td class="px-4 py-2 font-medium">${player.name} <span class="text-xs text-gray-500">(${player.role})</span> ${player.currentInningsOut ? '<span class="text-cricket-red font-bold">(OUT)</span>' : (player.currentInningsB > 0 ? '<span class="text-cricket-green font-bold">(NOT OUT)</span>' : '')}</td>
                    <td class="px-4 py-2 text-right">${player.currentInningsR}</td>
                    <td class="px-4 py-2 text-right">${player.currentInningsB}</td>
                    <td class="px-4 py-2 text-right">${player.currentInningsFours}</td>
                    <td class="px-4 py-2 text-right">${player.currentInningsSixes}</td>
                    <td class="px-4 py-2 text-right">${player.currentInningsB > 0 ? ((player.currentInningsR / player.currentInningsB) * 100).toFixed(2) : '0.00'}</td>
                </tr>
            `).join('');

            // Get bowling team's top bowler stats for the current match (accumulated)
            const bowlingTeamId = (gameState.currentBattingTeamId === gameState.userTeamId) ? gameState.compTeamId : gameState.userTeamId;
            const bowlers = gameState.teams[bowlingTeamId].players.filter(p => p.Balls > 0);
            const topBowlers = bowlers.sort((a, b) => b.W - a.W || a.Runs - b.Runs).slice(0, 3);
            const bowlingRows = topBowlers.map(bowler => {
                const totalBowledOversDecimal = bowler.O + (bowler.Balls % 6) / 6;
                const econ = (totalBowledOversDecimal > 0) ? (bowler.Runs / totalBowledOversDecimal).toFixed(2) : 0.0;
                return `
                    <tr class="bg-blue-50">
                        <td class="px-4 py-2 font-medium">${bowler.name} <span class="text-xs text-gray-500">(${bowler.role})</span></td>
                        <td class="px-4 py-2 text-right">${bowler.O}.${bowler.Balls % 6}</td>
                        <td class="px-4 py-2 text-right">${bowler.Runs}</td>
                        <td class="px-4 py-2 text-right">${bowler.W}</td>
                        <td class="px-4 py-2 text-right">${econ}</td>
                    </tr>
                `;
            }).join('');


            return `
                <div class="bg-white p-6 card mt-6">
                    <h3 class="text-xl font-bold mb-4 border-b pb-2 text-cricket-green">${battingTeam.name} Batting Scorecard (Innings ${gameState.innings})</h3>
                    <div class="overflow-x-auto">
                        <table class="min-w-full divide-y divide-gray-200">
                            <thead class="bg-gray-100">
                                <tr>
                                    <th class="px-4 py-2 text-left text-xs font-medium text-gray-500 uppercase tracking-wider">Batsman</th>
                                    <th class="px-4 py-2 text-right text-xs font-medium text-gray-500 uppercase tracking-wider">Runs</th>
                                    <th class="px-4 py-2 text-right text-xs font-medium text-gray-500 uppercase tracking-wider">Balls</th>
                                    <th class="px-4 py-2 text-right text-xs font-medium text-gray-500 uppercase tracking-wider">4s</th>
                                    <th class="px-4 py-2 text-right text-xs font-medium text-gray-500 uppercase tracking-wider">6s</th>
                                    <th class="px-4 py-2 text-right text-xs font-medium text-gray-500 uppercase tracking-wider">SR</th>
                                </tr>
                            </thead>
                            <tbody class="divide-y divide-gray-200">
                                ${tableRows || '<tr><td colspan="6" class="px-4 py-4 text-center text-gray-500">No balls bowled yet.</td></tr>'}
                            </tbody>
                        </table>
                    </div>

                    <h3 class="text-xl font-bold mt-6 mb-4 border-b pb-2 text-cricket-green">Bowling Analysis (Overall Match)</h3>
                    <div class="overflow-x-auto">
                        <table class="min-w-full divide-y divide-gray-200">
                            <thead class="bg-gray-100">
                                <tr>
                                    <th class="px-4 py-2 text-left text-xs font-medium text-gray-500 uppercase tracking-wider">Bowler</th>
                                    <th class="px-4 py-2 text-right text-xs font-medium text-gray-500 uppercase tracking-wider">Overs</th>
                                    <th class="px-4 py-2 text-right text-xs font-medium text-gray-500 uppercase tracking-wider">Runs</th>
                                    <th class="px-4 py-2 text-right text-xs font-medium text-gray-500 uppercase tracking-wider">Wickets</th>
                                    <th class="px-4 py-2 text-right text-xs font-medium text-gray-500 uppercase tracking-wider">Econ</th>
                                </tr>
                            </thead>
                            <tbody class="divide-y divide-gray-200">
                                ${bowlingRows.length > 0 ? bowlingRows : '<tr><td colspan="5" class="px-4 py-4 text-center text-gray-500">No bowling stats to show yet.</td></tr>'}
                            </tbody>
                        </table>
                    </div>
                </div>
            `;
        };

        // --- LEAGUE UI RENDERING ---
        
        /** Moves league to finished state. */
        const finishLeague = () => {
            // Ensure table is sorted before declaring winner (NRR is the tiebreaker)
            leagueState.table.sort((a, b) => 
                b.Pts - a.Pts || 
                b.W - a.W || 
                parseFloat(b.NRR) - parseFloat(a.NRR) || 
                a.name.localeCompare(b.name)
            );
            const winnerName = leagueState.table[0].name;
            if(confirm(`The World Super League is complete! ${winnerName} is the champion. Reset the league for a new season?`)) {
                leagueState.isActive = false;
                gameState.currentMode = 'career';
                saveState();
                navigateTo('start');
            }
        }
        
        /** Allows user to temporarily switch to Career Mode (Exhibition) while keeping League active. */
        const pauseLeagueAndGoHome = () => {
            gameState.currentMode = 'career'; // Switch context
            saveState();
            navigateTo('start');
        };

        const renderLeagueHome = () => {
            if (!leagueState.isActive) {
                return `<div class="max-w-xl mx-auto p-8 bg-white card text-center">
                    <h3 class="text-2xl font-bold text-cricket-red">League not active.</h3>
                    <p class="mb-4">Start a new season from the home menu.</p>
                    <button onclick="navigateTo('start')" class="btn-primary">Go Home</button>
                </div>`;
            }

            const nextMatchIndex = leagueState.fixtures.findIndex(f => f.status === 'upcoming');
            const liveMatchIndex = leagueState.fixtures.findIndex(f => f.status === 'live');
            
            // Determine the fixture to display/play. If one is live, display that.
            const currentFixtureIndex = liveMatchIndex !== -1 ? liveMatchIndex : nextMatchIndex;
            const currentFixture = currentFixtureIndex !== -1 ? leagueState.fixtures[currentFixtureIndex] : null;

            const nextMatchDisplay = currentFixture ?
                `${PRESET_TEAMS_DATA[currentFixture.home].name} vs ${PRESET_TEAMS_DATA[currentFixture.away].name}` :
                'League Complete!';

            const leagueFormatDisplay = leagueState.leagueFormat === 'test_match' ? 'Test Match' : `${leagueState.oversPerMatch} Overs`;

            const renderTable = () => {
                const userTeamName = gameState.teams[gameState.userTeamId].name;
                const userTeamKey = Object.keys(PRESET_TEAMS_DATA).find(key => PRESET_TEAMS_DATA[key].name === userTeamName);


                // Ensure the table is sorted correctly before rendering
                leagueState.table.sort((a, b) => 
                    b.Pts - a.Pts || 
                    b.W - a.W || 
                    parseFloat(b.NRR) - parseFloat(a.NRR) || 
                    a.name.localeCompare(b.name)
                );

                const tableRows = leagueState.table.map(t => `
                    <tr class="${t.teamKey === userTeamKey ? 'bg-yellow-100 font-bold' : 'bg-white'} border-b">
                        <td class="px-4 py-2">${t.name}</td>
                        <td class="px-4 py-2 text-center">${t.P}</td>
                        <td class="px-4 py-2 text-center text-green-600">${t.W}</td>
                        <td class="px-4 py-2 text-center text-red-600">${t.L}</td>
                        <td class="px-4 py-2 text-center">${t.T}</td>
                        <td class="px-4 py-2 text-right">${t.NRR}</td>
                        <td class="px-4 py-2 text-right font-extrabold">${t.Pts}</td>
                    </tr>
                `).join('');

                return `
                    <div class="overflow-x-auto card p-6 bg-white">
                        <h3 class="text-2xl font-bold mb-4 text-cricket-green border-b pb-2">League Standings</h3>
                        <table class="min-w-full divide-y divide-gray-200">
                            <thead class="bg-gray-100">
                                <tr>
                                    <th class="px-4 py-2 text-left text-xs font-medium text-gray-500 uppercase tracking-wider">Team</th>
                                    <th class="px-4 py-2 text-center text-xs font-medium text-gray-500 uppercase tracking-wider">P</th>
                                    <th class="px-4 py-2 text-center text-xs font-medium text-gray-500 uppercase tracking-wider">W</th>
                                    <th class="px-4 py-2 text-center text-xs font-medium text-gray-500 uppercase tracking-wider">L</th>
                                    <th class="px-4 py-2 text-center text-xs font-medium text-gray-500 uppercase tracking-wider">T/D</th>
                                    <th class="px-4 py-2 text-right text-xs font-medium text-gray-500 uppercase tracking-wider">NRR</th>
                                    <th class="px-4 py-2 text-right text-xs font-medium text-gray-500 uppercase tracking-wider">Pts</th>
                                </tr>
                            </thead>
                            <tbody class="divide-y divide-gray-200">
                                ${tableRows}
                            </tbody>
                        </table>
                    </div>
                `;
            };

            const renderFixtures = () => {
                 // Show the next 5 upcoming matches
                const upcomingFixtures = leagueState.fixtures.filter(f => f.status === 'upcoming').slice(0, 5);
                const completedFixtures = leagueState.fixtures.filter(f => f.status === 'completed').slice(-5).reverse(); // Show last 5 completed

                const renderFixtureRow = (f, isCompleted) => `
                    <tr class="${isCompleted ? 'bg-gray-50 text-gray-600' : (f.status === 'live' ? 'bg-red-200 font-semibold' : 'bg-white font-semibold')} border-b">
                        <td class="px-4 py-2">${PRESET_TEAMS_DATA[f.home].name}</td>
                        <td class="px-4 py-2 text-center">vs</td>
                        <td class="px-4 py-2">${PRESET_TEAMS_DATA[f.away].name}</td>
                        <td class="px-4 py-2 text-sm text-right">
                            ${isCompleted ? f.result.split('won')[0].trim() : (f.status === 'live' ? 'LIVE' : 'UPCOMING')}
                        </td>
                    </tr>
                `;

                return `
                    <div class="grid md:grid-cols-2 gap-6">
                        <div class="card p-6 bg-white">
                            <h3 class="text-xl font-bold mb-4 text-cricket-green border-b pb-2">Completed Matches (Latest)</h3>
                            <table class="min-w-full divide-y divide-gray-200">
                                <tbody>
                                    ${completedFixtures.map(f => renderFixtureRow(f, true)).join('')}
                                    ${completedFixtures.length === 0 ? '<tr><td colspan="4" class="p-4 text-center text-gray-500">No matches completed yet.</td></tr>' : ''}
                                </tbody>
                            </table>
                        </div>
                        <div class="card p-6 bg-white">
                            <h3 class="text-xl font-bold mb-4 text-cricket-green border-b pb-2">Upcoming Matches (Next 5)</h3>
                            <table class="min-w-full divide-y divide-gray-200">
                                <tbody>
                                    ${upcomingFixtures.map(f => renderFixtureRow(f, false)).join('')}
                                    ${upcomingFixtures.length === 0 ? '<tr><td colspan="4" class="p-4 text-center text-gray-500">All matches played!</td></tr>' : ''}
                                </tbody>
                            </table>
                        </div>
                    </div>
                `;
            };


            const buttonsHtml = currentFixture ? `
                <button onclick="setupLeagueMatch(leagueState.fixtures[${currentFixtureIndex}])" class="btn-primary w-full text-xl mb-4">
                    ${currentFixture.status === 'live' ? 'Resume Current Match: ' : 'Play Next Match: '}
                    ${nextMatchDisplay}
                </button>
            ` : `
                <button onclick="finishLeague()" class="btn-primary w-full text-xl bg-cricket-gold text-cricket-green hover:bg-yellow-500 mb-4">
                    League Completed - View Final Table
                </button>
            `;

            return `
                <div class="max-w-6xl mx-auto">
                    <h2 class="text-4xl font-extrabold text-center mb-4 text-purple-700">World Super League</h2>
                    <p class="text-center text-xl mb-8 text-gray-700">Format: ${leagueFormatDisplay} | Matches Remaining: ${leagueState.fixtures.filter(f => f.status === 'upcoming' || f.status === 'live').length}</p>

                    ${buttonsHtml}
                    
                    ${renderTable()}

                    <div class="mt-8">
                        ${renderFixtures()}
                    </div>
                    
                    <div class="mt-8 text-center space-x-4">
                        <button onclick="navigateTo('league_records')" class="btn-primary bg-blue-600 hover:bg-blue-700">View League Stats</button>
                        <button onclick="pauseLeagueAndGoHome()" class="btn-primary bg-gray-600 hover:bg-gray-700">Go to Home (Exhibition Match)</button>
                        <button onclick="if(confirm('Are you sure you want to quit the current league season? All league progress will be lost.')) { leagueState.isActive = false; gameState.currentMode = 'career'; saveState(); navigateTo('start'); }" class="btn-primary bg-cricket-red hover:bg-red-700">Quit League and Reset</button>
                    </div>
                </div>
            `;
        };

        // --- MATCH MANAGEMENT LOGIC ---

        /** Sets up the game state for a new career match. */
        const startNewMatch = () => {
            clearMatchScoreboardState();
            gameState.currentMode = 'career';
            gameState.oversPerInnings = (gameState.gameMode === 'test_match') ? TEST_MATCH_OVERS : gameState.selectedOvers;
            gameState.isMatchActive = true; 

            // Load preset teams based on selection (if user changed them but hasn't updated names)
            loadPresetTeam('A', gameState.selectedPresetTeamA);
            loadPresetTeam('B', gameState.selectedPresetTeamB);

            navigateTo('toss');
        };

        /** NEW: Helper to fully clear the volatile state of a match (scores, innings history, results) */
        const clearMatchScoreboardState = () => {
            gameState.isMatchActive = false; 
            gameState.tossWinner = null;
            gameState.tossDecision = null;
            gameState.target = 0;
            gameState.innings = 1;
            gameState.matchResult = null;
            gameState.summary = null;
            gameState.testMatchState = getInitialGameState().testMatchState;
            gameState.score = getInitialGameState().score;
            
            // Clear all accumulated stats for the match (R, W, etc. within the player object)
            // Ensure the match players are reset to their base stat structure, keeping name/role.
            ['A', 'B'].forEach(teamId => {
                const currentPlayers = gameState.teams[teamId].players.map(p => ({name: p.name, role: p.role}));
                gameState.teams[teamId].players = currentPlayers.map(p => getNewPlayerStats(p.name, p.role));
                
                gameState.teams[teamId].inningsScores = [];
                gameState.teams[teamId].inningsScorecardHistory = [];
            });
            saveState(); // Save the cleared state
        };
        
        /** Sets up the game state for a new league match based on a fixture. */
        const setupLeagueMatch = (fixture) => {
            const teamAKey = fixture.home;
            const teamBKey = fixture.away;
            
            const teamAData = PRESET_TEAMS_DATA[teamAKey];
            const teamBData = PRESET_TEAMS_DATA[teamBKey];
            
            // If starting a brand new fixture, clear the scoresheet
            if (fixture.status === 'upcoming') {
                clearMatchScoreboardState(); 
            }

            gameState.currentMode = 'league';
            
            // 2. Apply League settings
            gameState.gameMode = leagueState.leagueFormat;
            gameState.oversPerInnings = leagueState.oversPerMatch;
            
            // 3. Set Team A/B rosters and names based on fixture
            gameState.teams['A'].name = teamAData.name;
            // When setting players for a league match, load the preset roster but preserve match stats if resuming a match
            gameState.teams['A'].players = teamAData.players.map(p => {
                // Find existing player match totals if resuming, otherwise start fresh
                const existingPlayer = gameState.teams['A'].players.find(existing => existing.name === p.name);
                const newPlayer = getNewPlayerStats(p.name, p.role);
                // If resuming a live match, preserve match totals (R, B, W, etc.) but reset current innings
                if (fixture.status === 'live' && existingPlayer) {
                    newPlayer.R = existingPlayer.R;
                    newPlayer.B = existingPlayer.B;
                    newPlayer.W = existingPlayer.W;
                    newPlayer.Runs = existingPlayer.Runs;
                    newPlayer.O = existingPlayer.O;
                    newPlayer.Balls = existingPlayer.Balls;
                }
                return newPlayer;
            });
            
            gameState.teams['B'].name = teamBData.name;
            gameState.teams['B'].players = teamBData.players.map(p => {
                 const existingPlayer = gameState.teams['B'].players.find(existing => existing.name === p.name);
                 const newPlayer = getNewPlayerStats(p.name, p.role);
                 if (fixture.status === 'live' && existingPlayer) {
                    newPlayer.R = existingPlayer.R;
                    newPlayer.B = existingPlayer.B;
                    newPlayer.W = existingPlayer.W;
                    newPlayer.Runs = existingPlayer.Runs;
                    newPlayer.O = existingPlayer.O;
                    newPlayer.Balls = existingPlayer.Balls;
                 }
                 return newPlayer;
            });
            
            // 4. Assign user team based on preset preference (if user team is in fixture)
            const userLeagueTeamName = gameState.teams[gameState.userTeamId].name;
            const userLeagueTeamKey = Object.keys(PRESET_TEAMS_DATA).find(key => PRESET_TEAMS_DATA[key].name === userLeagueTeamName);
            
            if (teamAKey === userLeagueTeamKey) {
                gameState.userTeamId = 'A';
                gameState.compTeamId = 'B';
            } else if (teamBKey === userLeagueTeamKey) {
                gameState.userTeamId = 'B';
                gameState.compTeamId = 'A';
            } else {
                // If user is not playing, default to A vs B
                gameState.userTeamId = 'A';
                gameState.compTeamId = 'B';
            }

            // 5. Update fixture status and index
            const indexToUpdate = leagueState.fixtures.findIndex(f => f.home === teamAKey && f.away === teamBKey);
            if (indexToUpdate !== -1) {
                leagueState.currentMatchIndex = indexToUpdate;
                leagueState.fixtures[indexToUpdate].status = 'live';
            } 
            
            saveState();
            navigateTo('toss');
        };

        /** Updates the selected overs in gameState */
        const updateSelectedOvers = (value) => {
            gameState.selectedOvers = parseInt(value, 10);
            gameState.oversPerInnings = parseInt(value, 10); // Update actual overs for current mode if limited overs
            saveState();
            renderScreen(); // Re-render to update the button text
        };

        /** Sets the game mode (limited overs or test match). */
        const setGameMode = (mode) => {
            gameState.gameMode = mode;
            gameState.oversPerInnings = (mode === 'test_match') ? TEST_MATCH_OVERS : gameState.selectedOvers;
            saveState();
            renderScreen(); // Re-render to show/hide overs selection
        };

        // NEW: Helper function to setup bowling indices for a new innings
        const setupNewInningsBowling = () => {
            const bowlingTeamId = (gameState.currentBattingTeamId === gameState.userTeamId) ? gameState.compTeamId : gameState.userTeamId;
            const bowlingTeam = gameState.teams[bowlingTeamId];

            // Find the lowest index player who is an eligible bowler
            let initialBowlerIndex = bowlingTeam.players.findIndex(p => ELIGIBLE_BOWLERS.includes(p.role));
            
            if (initialBowlerIndex === -1) {
                // Fallback to index 5 if no specialized bowler exists, but this means batters/keepers will be bowling.
                initialBowlerIndex = 5; 
                console.warn(`No dedicated bowler/all-rounder found for ${bowlingTeam.name}. Falling back to default index.`);
            }

            // Apply the selection
            gameState.score.bowling.currentBowlerIndex = initialBowlerIndex;
            gameState.score.bowling.lastBowlerIndex = -1;
        };


        /** Performs the initial toss and sets up the first innings. */
        const performToss = () => {
            const winner = Math.random() < 0.5 ? 'user' : 'comp';
            gameState.tossWinner = winner;

            if (winner === 'comp') {
                gameState.tossDecision = Math.random() < 0.5 ? 'bat' : 'bowl';
                gameState.isUserBatting = (gameState.tossDecision === 'bowl'); // If comp bowls, user bats
                gameState.currentBattingTeamId = gameState.isUserBatting ? gameState.userTeamId : gameState.compTeamId;
            }
            // If user wins, decision is deferred until userChooseBat/Bowl

            if (gameState.tossDecision) { // If a decision has been made
                // Setup team assignments for Test Match
                if (gameState.gameMode === 'test_match') {
                    if (gameState.isUserBatting) {
                        gameState.testMatchState.team1Id = gameState.userTeamId;
                        gameState.testMatchState.team2Id = gameState.compTeamId;
                    } else {
                        gameState.testMatchState.team1Id = gameState.compTeamId;
                        gameState.testMatchState.team2Id = gameState.userTeamId;
                    }
                }
                // Setup the initial bowler dynamically
                setupNewInningsBowling();
            }
            
            gameState.isMatchActive = true; 
            saveState();
            renderScreen();
        };

        /** User chooses to bat after winning the toss. */
        const userChooseBat = () => {
            gameState.tossDecision = 'bat';
            gameState.isUserBatting = true;
            gameState.currentBattingTeamId = gameState.userTeamId; // User bats first
            if (gameState.gameMode === 'test_match') {
                gameState.testMatchState.team1Id = gameState.userTeamId;
                gameState.testMatchState.team2Id = gameState.compTeamId;
            }
            gameState.isMatchActive = true; 
            setupNewInningsBowling(); // Set initial bowler
            saveState();
            renderScreen();
        };

        /** User chooses to bowl after winning the toss. */
        const userChooseBowl = () => {
            gameState.tossDecision = 'bowl';
            gameState.isUserBatting = false;
            gameState.currentBattingTeamId = gameState.compTeamId; // Computer bats first
            if (gameState.gameMode === 'test_match') {
                gameState.testMatchState.team1Id = gameState.compTeamId;
                gameState.testMatchState.team2Id = gameState.userTeamId;
            }
            gameState.isMatchActive = true; 
            setupNewInningsBowling(); // Set initial bowler
            saveState();
            renderScreen();
        };

        /** Selects the next bowler based on role and overs bowled. */
        const selectNextBowler = () => {
            const bowlingTeamId = (gameState.currentBattingTeamId === gameState.userTeamId) ? gameState.compTeamId : gameState.userTeamId;
            const bowlingTeam = gameState.teams[bowlingTeamId];
            const lastBowlerIndexForRotation = gameState.score.bowling.currentBowlerIndex;

            const eligibleBowlersInfo = [];
            bowlingTeam.players.forEach((player, index) => {
                // STRICTLY: Only allow Bowler and All Rounder roles to bowl
                if (ELIGIBLE_BOWLERS.includes(player.role)) {
                    eligibleBowlersInfo.push({ index: index, player: player, overs: player.O });
                }
            });

            if (eligibleBowlersInfo.length === 0) {
                // If no eligible bowlers, the previous bowler must continue (even if they are a Batter/WK by mistake)
                console.warn("No eligible Bowler/All Rounder found. Previous bowler continuing.");
                return;
            }

            // Sort eligible bowlers: 1. By least overs bowled, 2. By index (to ensure fair rotation if overs are tied)
            eligibleBowlersInfo.sort((a, b) => a.overs - b.overs || a.index - b.index);

            let nextBowlerIndex = -1;
            
            // 1. Find the best available bowler who DIDN'T bowl the last over.
            for (const bowlerInfo of eligibleBowlersInfo) {
                if (bowlerInfo.index !== lastBowlerIndexForRotation) {
                    nextBowlerIndex = bowlerInfo.index;
                    break;
                }
            }

            // 2. If no available bowler is found (e.g., only one bowler, or all others have bowled the same number as the last one),
            // or if the loop finished without finding an alternative, just pick the top available bowler (lowest overs/index).
            if (nextBowlerIndex === -1 && eligibleBowlersInfo.length > 0) {
                 // Re-select the top bowler (could be the previous bowler if no one else is viable, e.g., one bowler team)
                nextBowlerIndex = eligibleBowlersInfo[0].index; 
            } 

            // Apply the selected bowler index
            if (nextBowlerIndex !== -1) {
                gameState.score.bowling.lastBowlerIndex = lastBowlerIndexForRotation;
                gameState.score.bowling.currentBowlerIndex = nextBowlerIndex;
            }
        };


        /** Handles the logic for a single ball (dice roll). Returns true if innings ended, false otherwise. */
        const handleRoll = async (manualBatRoll = null) => {
            if (gameState.score.wickets === 10) return checkInningsEnd(); // Already all out
            
            const isUserBatting = gameState.currentBattingTeamId === gameState.userTeamId;
            let batRoll;

            // Determine the Batsman's Roll
            if (manualBatRoll !== null && isUserBatting && !gameState.isSimulating) {
                batRoll = manualBatRoll; // Use the value chosen by the player
            } else {
                batRoll = rollDice(); // Random roll for simulation or computer turn
            }
            
            // Bowler's Roll is always random
            const bowlRoll = rollDice();
            
            const isOut = batRoll === bowlRoll;
            const runsScored = isOut ? 0 : batRoll;

            // Dice animation visualization - only if NOT simulating
            if (!gameState.isSimulating) {
                const batDiceContainer = document.getElementById('batting-dice');
                const bowlDiceContainer = document.getElementById('bowling-dice');

                if (batDiceContainer && bowlDiceContainer) {
                    batDiceContainer.innerHTML = getDiceHtml(1);
                    bowlDiceContainer.innerHTML = getDiceHtml(1);
                    batDiceContainer.classList.add('roll-animation');
                    bowlDiceContainer.classList.add('roll-animation');
                }
                // Wait for animation
                await new Promise(resolve => setTimeout(resolve, 50)); // Reduced animation delay for quicker response
                if (batDiceContainer && bowlDiceContainer) {
                    batDiceContainer.classList.remove('roll-animation');
                    bowlDiceContainer.classList.remove('roll-animation');
                    batDiceContainer.innerHTML = getDiceHtml(batRoll);
                    bowlDiceContainer.innerHTML = getDiceHtml(bowlRoll);
                }
            }


            updateScore(isOut, runsScored);

            const actionMessage = document.getElementById('action-message');
            if (actionMessage) {
                if (isOut) {
                    actionMessage.innerHTML = `<span class="font-bold text-cricket-red">OUT!</span> Batsman rolled ${batRoll}, Bowler rolled ${bowlRoll}.`;
                } else {
                    actionMessage.innerHTML = `<span class="font-bold text-cricket-green">Run!</span> Batsman rolled ${batRoll}, Bowler rolled ${bowlRoll}. Scores ${runsScored}.`;
                }
            }

            const inningsEnded = checkInningsEnd();
            
            // If the innings hasn't ended, we must manually re-render the screen
            if (!inningsEnded && !gameState.isSimulating) {
                renderScreen();
            }
            
            return inningsEnded;
        };

        /** Handles the manual dice selection by the user when batting. */
        const handleManualRoll = (batRollValue) => {
            handleRoll(batRollValue);
        };
        
        /** Checks if the innings or match has ended. Returns true if innings ended, false otherwise. */
        const checkInningsEnd = () => {
            const score = gameState.score.batting;

            const allOut = score.wickets === 10;
            const oversComplete = score.overs === gameState.oversPerInnings;

            let targetChased = false;
            if (gameState.target > 0 && score.runs >= gameState.target) {
                if (gameState.gameMode === 'limited_overs' && gameState.innings === 2) {
                    targetChased = true;
                } else if (gameState.gameMode === 'test_match' && gameState.innings === 4) {
                    targetChased = true;
                }
            }

            if (allOut || oversComplete || targetChased) {
                // Ensure last batsman is marked as having played in scorecard slice
                if (score.nextPlayerIndex <= 11) score.nextPlayerIndex = 11;
                endInnings(); // This will trigger navigateTo('summary') or showInningsBreakMessage()
                return true; // Indicate innings ended
            }
            return false; // Indicate innings did not end
        };

        /** Updates the central game score based on the dice outcome and handles two batsmen logic. */
        const updateScore = (isOut, runsScored) => {
            const score = gameState.score.batting;
            const battingTeam = gameState.teams[gameState.currentBattingTeamId];
            const bowlingTeamId = (gameState.currentBattingTeamId === gameState.userTeamId) ? gameState.compTeamId : gameState.userTeamId;
            const bowlingTeam = gameState.teams[bowlingTeamId];

            const currentBatsmanOnStrike = battingTeam.players[score.onStrikePlayerIndex];
            const bowler = bowlingTeam.players[gameState.score.bowling.currentBowlerIndex];

            // Update Current Innings Stats for Batsman
            currentBatsmanOnStrike.currentInningsB += 1;
            currentBatsmanOnStrike.currentInningsR += runsScored;
            if (runsScored === 4) currentBatsmanOnStrike.currentInningsFours += 1;
            if (runsScored === 6) currentBatsmanOnStrike.currentInningsSixes += 1;
            currentBatsmanOnStrike.currentInningsSR = currentBatsmanOnStrike.currentInningsB > 0 ? ((currentBatsmanOnStrike.currentInningsR / currentBatsmanOnStrike.currentInningsB) * 100).toFixed(2) : 0.0;

            // Update Match Total Stats for Batsman (for career records)
            currentBatsmanOnStrike.B += 1;
            currentBatsmanOnStrike.R += runsScored;
            if (runsScored === 4) currentBatsmanOnStrike.Fours += 1;
            if (runsScored === 6) currentBatsmanOnStrike.Sixes += 1;

            // Update Bowler's Match Total Stats
            bowler.Balls += 1;
            bowler.Runs += runsScored;

            // Update Team Score for current innings
            score.runs += runsScored;

            // Handle Wicket
            if (isOut) {
                currentBatsmanOnStrike.currentInningsOut = true; // Set current innings out status
                score.wickets += 1;
                bowler.W += 1;

                // Bring in next player if available
                if (score.wickets < 10) {
                    score.onStrikePlayerIndex = score.nextPlayerIndex; // New batsman takes strike
                    score.nextPlayerIndex++;
                }
            } else {
                if (runsScored % 2 !== 0 && runsScored > 0) {
                    [score.onStrikePlayerIndex, score.offStrikePlayerIndex] = [score.offStrikePlayerIndex, score.onStrikePlayerIndex];
                }
            }

            // Update Ball Count for the overall innings
            score.balls += 1;

            // Check for End of Over
            if (score.balls === 6) {
                score.overs += 1;
                score.balls = 0;

                // Swap strike at the end of the over
                [score.onStrikePlayerIndex, score.offStrikePlayerIndex] = [score.offStrikePlayerIndex, score.onStrikePlayerIndex];

                // Change Bowler
                selectNextBowler();
            }

            // Update Bowler's match total Overs and Econ
            bowler.O = Math.floor(bowler.Balls / 6);
            const totalBowledOversDecimal = bowler.O + (bowler.Balls % 6) / 6;

            if (totalBowledOversDecimal > 0) {
                bowler.Econ = (bowler.Runs / totalBowledOversDecimal).toFixed(2);
            } else {
                bowler.Econ = 0.0;
            }

            saveState();
        };

        /** Finalizes the current innings score and prepares for the next. */
        const endInnings = () => {
            const currentBattingTeam = gameState.teams[gameState.currentBattingTeamId];
            const score = gameState.score.batting;

            console.log(`End of Innings (current game.innings: ${gameState.innings}). Team: ${currentBattingTeam.name}. Score: ${score.runs}/${score.wickets}`);

            // Snapshot player stats for the completed innings
            const inningsPlayersSnapshot = currentBattingTeam.players.slice(0, score.nextPlayerIndex).map(p => ({
                name: p.name,
                role: p.role,
                R: p.currentInningsR,
                B: p.currentInningsB,
                Fours: p.currentInningsFours,
                Sixes: p.currentInningsSixes,
                SR: p.currentInningsSR,
                out: p.currentInningsOut
            }));
            currentBattingTeam.inningsScorecardHistory.push(inningsPlayersSnapshot);

            // Store final innings score
            currentBattingTeam.inningsScores.push({
                runs: score.runs,
                wickets: score.wickets,
                overs: score.overs,
                balls: score.balls,
                inningsNum: gameState.innings
            });


            if (gameState.gameMode === 'limited_overs') {
                handleLimitedOversInningsEnd();
            } else { // Test Match
                handleTestMatchInningsEnd();
            }
        };

        const handleLimitedOversInningsEnd = () => {
            if (gameState.innings === 2) {
                endMatch();
                return;
            }

            // Prepare for Innings 2
            gameState.innings = 2;
            gameState.target = gameState.score.batting.runs + 1;
            gameState.isUserBatting = !gameState.isUserBatting;
            gameState.currentBattingTeamId = gameState.isUserBatting ? gameState.userTeamId : gameState.compTeamId;

            resetCurrentInningsScoreState();
            setupNewInningsBowling(); // Initialize bowler for new innings
            
            // Only show break message if not simulating fully
            if (!gameState.isSimulating) {
                showInningsBreakMessage();
            }
        };

        const handleTestMatchInningsEnd = () => {
            const team1Id = gameState.testMatchState.team1Id;
            const team2Id = gameState.testMatchState.team2Id;

            // Scenario 1: After Innings 2 (Team 2's 1st innings) - Potential win by an innings for Team 1
            if (gameState.innings === 2) {
                const team1Innings1Score = gameState.teams[team1Id].inningsScores[0]?.runs || 0;
                const team2Innings1Score = gameState.teams[team2Id].inningsScores[0]?.runs || 0; 

                if (team2Innings1Score < team1Innings1Score) {
                    // Check for follow-on possibility (usually 200 runs lead in 5-day tests, but here we just check if Team 2 trailed)
                    if ((team1Innings1Score - team2Innings1Score) > 0) {
                        gameState.testMatchState.matchFinishedByInningsWin = true;
                        gameState.matchResult = `${gameState.teams[team1Id].name} won by an innings and ${team1Innings1Score - team2Innings1Score} runs!`;
                        endMatch();
                        return;
                    }
                }
            }

            // Scenario 2: After Innings 3 (Team 1's 2nd innings) - Potential win by an innings for Team 2
            if (gameState.innings === 3) {
                const team1TotalRuns = (gameState.teams[team1Id].inningsScores[0]?.runs || 0) + (gameState.teams[team1Id].inningsScores[1]?.runs || 0);
                const team2FirstInningsRuns = (gameState.teams[team2Id].inningsScores[0]?.runs || 0);

                if (team1TotalRuns <= team2FirstInningsRuns) {
                    gameState.testMatchState.matchFinishedByInningsWin = true;
                    gameState.matchResult = `${gameState.teams[team2Id].name} won by an innings and ${team2FirstInningsRuns - team1TotalRuns} runs!`;
                    endMatch();
                    return;
                }
            }

            // Scenario 3: After Innings 4 (Team 2's 2nd innings) - Final match conclusion
            if (gameState.innings === 4) {
                endMatch();
                return;
            }

            // --- If match not ended, prepare for next innings ---
            gameState.innings++; // Progress to the next innings (will be 2, 3, or 4)

            // Reset current innings score state for the next batting team
            resetCurrentInningsScoreState();

            // Determine next batting team and set `isUserBatting` accordingly
            const battingOrder = [team1Id, team2Id, team1Id, team2Id]; // Maps overall innings (1-indexed) to team IDs
            gameState.currentBattingTeamId = battingOrder[gameState.innings - 1]; // Use 0-indexed array
            gameState.isUserBatting = (gameState.currentBattingTeamId === gameState.userTeamId);
            
            setupNewInningsBowling(); // Initialize bowler for new innings

            // Calculate target for Innings 4 if we are moving to it
            if (gameState.innings === 4) {
                const team1TotalRuns = (gameState.teams[team1Id].inningsScores[0]?.runs || 0) + (gameState.teams[team1Id].inningsScores[1]?.runs || 0);
                const team2FirstInningsRuns = (gameState.teams[team2Id].inningsScores[0]?.runs || 0);
                gameState.target = team1TotalRuns - team2FirstInningsRuns + 1;
                if (gameState.target <= 0) gameState.target = 1; // Safety net: target must be at least 1
            } else {
                gameState.target = 0; // No target for innings 1, 2, 3
            }

            // Update lead/trail message
            gameState.testMatchState.currentLeadTrailMessage = calculateTestLeadTrailMessage();
            // Only show break message if not simulating fully
            if (!gameState.isSimulating) {
                showInningsBreakMessage();
            }
        };

        /** Resets the current innings' score trackers and batsman's current innings stats. */
        const resetCurrentInningsScoreState = () => {
            gameState.score.batting = {
                runs: 0, wickets: 0, overs: 0, balls: 0,
                onStrikePlayerIndex: 0, offStrikePlayerIndex: 1, nextPlayerIndex: 2
            };
            // Note: Bowling state will be reset and setup by setupNewInningsBowling after this is called.
            gameState.score.bowling = {
                overs: 0, balls: 0, wickets: 0, runsConceded: 0,
                currentBowlerIndex: 5, // Placeholder
                lastBowlerIndex: -1
            };

            // Reset player-specific CURRENT INNINGS stats for the team that is about to bat
            const currentBattingTeam = gameState.teams[gameState.currentBattingTeamId];
            currentBattingTeam.players.forEach(p => {
                p.currentInningsR = 0; p.currentInningsB = 0; p.currentInningsFours = 0;
                p.currentInningsSixes = 0; p.currentInningsSR = 0.0; p.currentInningsOut = false;
            });
            // Note: Match-total bowling stats (W, Runs, etc.) are NOT reset here; they accumulate across the match.
        };

        /** Calculates and returns the lead/trail message for Test matches. */
        const calculateTestLeadTrailMessage = () => {
            const team1Id = gameState.testMatchState.team1Id;
            const team2Id = gameState.testMatchState.team2Id;

            const team1Scores = gameState.teams[team1Id]?.inningsScores || [];
            const team2Scores = gameState.teams[team2Id]?.inningsScores || [];

            let message = "";

            if (gameState.innings === 2) { // During Team 2's 1st innings
                if (team1Scores.length > 0) { // If Team 1 has completed its first innings
                    const team1Innings1 = team1Scores[0].runs;
                    const team2CurrentScore = gameState.score.batting.runs;
                    const diff = team1Innings1 - team2CurrentScore;
                    if (diff > 0) {
                        message = `${gameState.teams[team2Id].name} trails by ${diff} runs.`;
                    } else if (diff < 0) {
                        message = `${gameState.teams[team2Id].name} leads by ${Math.abs(diff)} runs.`;
                    } else {
                        message = `Scores are level.`;
                    }
                }
            } else if (gameState.innings === 3) { // After Team 2's 1st innings, during Team 1's 2nd innings
                const team1Innings1 = team1Scores[0]?.runs || 0;
                const team2Innings1 = team2Scores[0]?.runs || 0;
                const team1CurrentInnings2Score = gameState.score.batting.runs; // Team 1's score in their 2nd innings

                const team1TotalSoFar = team1Innings1 + team1CurrentInnings2Score;
                const diff = team1TotalSoFar - team2Innings1;

                if (diff > 0) {
                    message = `${gameState.teams[team1Id].name} leads by ${diff} runs.`;
                } else if (diff < 0) {
                    message = `${gameState.teams[team1Id].name} trails by ${Math.abs(diff)} runs.`; // Should be rare
                } else {
                    message = `Scores are level cumulatively.`;
                }
            } else if (gameState.innings === 4) { // After Team 1's 2nd innings, during Team 2's 2nd innings
                const team1TotalRuns = (team1Scores[0]?.runs || 0) + (team1Scores[1]?.runs || 0);
                const team2Innings1 = (team2Scores[0]?.runs || 0);
                const target = team1TotalRuns - team2Innings1 + 1;

                // Message for chasing team
                if (target > 0) {
                    const team2CurrentScore = gameState.score.batting.runs;
                    const runsNeeded = target - team2CurrentScore;
                    if (runsNeeded > 0) {
                        message = `${gameState.teams[team2Id].name} needs ${runsNeeded} runs to win.`;
                    } else {
                        message = `${gameState.teams[team2Id].name} has reached the target!`; 
                    }
                }
            }
            return message;
        };


        /** Displays a modal/message for the innings break. */
        const showInningsBreakMessage = () => {
            let messageTitle = "INNINGS BREAK!";
            let mainMessage = "";
            let continueButtonText = "Continue";

            // If a match result is already set, don't show innings break, go to summary
            if (gameState.matchResult) {
                navigateTo('summary');
                return;
            }

            if (gameState.gameMode === 'limited_overs') {
                const innings1Score = gameState.teams[gameState.currentBattingTeamId === 'A' ? 'B' : 'A'].inningsScores[0];
                const innings1TeamName = gameState.teams[gameState.currentBattingTeamId === 'A' ? 'B' : 'A'].name;
                const innings2TeamName = gameState.teams[gameState.currentBattingTeamId].name;
                
                // Re-calculate target based on actual runs scored in Innings 1
                const target = (innings1Score?.runs || 0) + 1; 

                mainMessage = `End of Innings 1. Target set! ${innings1TeamName} scored ${target - 1} runs.
                                <br><strong class="text-cricket-green">${innings2TeamName} will now bat in Innings 2. They need ${target} runs to win from ${gameState.oversPerInnings} overs.</strong>`;
                continueButtonText = "Continue to Innings 2";
            } else { // Test Match
                const team1Id = gameState.testMatchState.team1Id;
                const team2Id = gameState.testMatchState.team2Id;

                const team1Name = gameState.teams[team1Id].name;
                const team2Name = gameState.teams[team2Id].name;

                let nextBattingTeamName = '';
                if (gameState.innings === 2) nextBattingTeamName = team2Name;
                else if (gameState.innings === 3) nextBattingTeamName = team1Name;
                else if (gameState.innings === 4) nextBattingTeamName = team2Name;

                messageTitle = `End of Innings ${gameState.innings - 1}!`; // Innings has already been incremented by now
                if (gameState.innings === 2) {
                    mainMessage = `<strong>${team1Name} completed their 1st innings.</strong>`;
                    mainMessage += `<br><br> ${nextBattingTeamName} will now bat in Innings ${gameState.innings}.`;
                } else if (gameState.innings === 3) {
                    mainMessage = `<strong>${team2Name} completed their 1st innings.</strong>`;
                    if (gameState.testMatchState.currentLeadTrailMessage) {
                        mainMessage += `<br>${gameState.testMatchState.currentLeadTrailMessage}`;
                    }
                    mainMessage += `<br><br> ${nextBattingTeamName} will now bat in Innings ${gameState.innings}.`;
                } else if (gameState.innings === 4) {
                    mainMessage = `<strong>${team1Name} completed their 2nd innings.</strong>`;
                     if (gameState.testMatchState.currentLeadTrailMessage) {
                        mainMessage += `<br>${gameState.testMatchState.currentLeadTrailMessage}`;
                    }
                    mainMessage += `<br><br> ${nextBattingTeamName} will now bat in Innings ${gameState.innings} (Final Chase). They need ${gameState.target} runs to win.`;
                    continueButtonText = "Continue to Final Innings";
                }
            }

            const message = `
                <div id="innings-break-modal" class="fixed inset-0 bg-black bg-opacity-70 flex items-center justify-center z-50 p-4">
                    <div class="bg-white card p-8 w-full max-w-md text-center">
                        <h3 class="text-3xl font-bold mb-4 text-cricket-gold">${messageTitle}</h3>
                        <p class="text-xl mb-6">${mainMessage}</p>
                        <button onclick="document.getElementById('innings-break-modal').remove(); navigateTo('match');" class="btn-primary">
                            ${continueButtonText}
                        </button>
                    </div>
                </div>
            `;
            document.body.insertAdjacentHTML('beforeend', message);
        };

        /** Determines the match winner and sets the final result. */
        const endMatch = () => {
            let resultLine = '';
            let winnerTeamId = '';
            let winnerTeamName = '';

            const teamA = gameState.teams['A'];
            const teamB = gameState.teams['B'];

            if (gameState.gameMode === 'limited_overs') {
                const firstBattingTeamId = (gameState.tossDecision === 'bat' && gameState.tossWinner === 'user') || (gameState.tossDecision === 'bowl' && gameState.tossWinner === 'comp') ? gameState.userTeamId : gameState.compTeamId;
                const secondBattingTeamId = (firstBattingTeamId === gameState.userTeamId) ? gameState.compTeamId : gameState.userTeamId;

                const innings1Score = gameState.teams[firstBattingTeamId].inningsScores[0];
                const innings2Score = gameState.teams[secondBattingTeamId].inningsScores[0];

                if (!innings1Score || !innings2Score) {
                    resultLine = "Error: Incomplete scores";
                } else if (innings2Score.runs >= gameState.target) {
                    winnerTeamId = secondBattingTeamId;
                    winnerTeamName = gameState.teams[winnerTeamId].name;
                    const wicketsRemaining = 10 - innings2Score.wickets;
                    resultLine = `${winnerTeamName} won by ${wicketsRemaining} wickets`;
                } else if (innings1Score.runs > innings2Score.runs) {
                    winnerTeamId = firstBattingTeamId;
                    winnerTeamName = gameState.teams[winnerTeamId].name;
                    const runMargin = innings1Score.runs - innings2Score.runs;
                    resultLine = `${winnerTeamName} won by ${runMargin} runs`;
                } else {
                    resultLine = `The match was tied!`;
                    winnerTeamName = "No Winner";
                }
            } else { // Test Match
                const team1Id = gameState.testMatchState.team1Id;
                const team2Id = gameState.testMatchState.team2Id;
                const team1Name = gameState.teams[team1Id].name;
                const team2Name = gameState.teams[team2Id].name;

                if (gameState.testMatchState.matchFinishedByInningsWin) {
                    resultLine = gameState.matchResult;
                    winnerTeamName = (resultLine.includes(team1Name)) ? team1Name : team2Name;
                    winnerTeamId = (resultLine.includes(team1Name)) ? team1Id : team2Id;
                } else {
                    const team1TotalRuns = (gameState.teams[team1Id].inningsScores[0]?.runs || 0) + (gameState.teams[team1Id].inningsScores[1]?.runs || 0);
                    const team2TotalRunsInnings1 = (gameState.teams[team2Id].inningsScores[0]?.runs || 0);
                    const team2CurrentInnings4Score = gameState.score.batting.runs; 

                    if (gameState.innings === 4 && team2CurrentInnings4Score >= gameState.target) {
                        winnerTeamId = team2Id;
                        winnerTeamName = team2Name;
                        const wicketsRemaining = 10 - gameState.score.batting.wickets;
                        resultLine = `${winnerTeamName} won by ${wicketsRemaining} wickets`;
                    }
                    else if (gameState.innings === 4 && team2CurrentInnings4Score < gameState.target && gameState.score.batting.wickets === 10) {
                        winnerTeamId = team1Id;
                        winnerTeamName = team1Name;
                        const finalRunDifference = (team1TotalRuns - team2TotalRunsInnings1) - team2CurrentInnings4Score; 
                        resultLine = `${winnerTeamName} won by ${finalRunDifference} runs`;
                    }
                    else {
                        // Match ended by overs/time in the 4th innings without a result being reached
                        resultLine = `The match was drawn!`;
                        winnerTeamName = "No Winner";
                        winnerTeamId = null;
                    }
                }
            }
            gameState.matchResult = resultLine;
            gameState.isMatchActive = false;
            gameState.isSimulating = false; // Ensure simulation stops

            // 1. Calculate Man of the Match and final summary structure
            const firstBattingOverallTeamId = gameState.gameMode === 'test_match' ? gameState.testMatchState.team1Id : ((gameState.tossDecision === 'bat' && gameState.tossWinner === 'user') || (gameState.tossDecision === 'bowl' && gameState.tossWinner === 'comp') ? gameState.userTeamId : gameState.compTeamId);
            const secondBattingOverallTeamId = gameState.gameMode === 'test_match' ? gameState.testMatchState.team2Id : (firstBattingOverallTeamId === gameState.userTeamId ? gameState.compTeamId : gameState.userTeamId);

            const summary = generateMatchSummary(winnerTeamName, gameState.matchResult, firstBattingOverallTeamId, secondBattingOverallTeamId);
            gameState.summary = summary;

            // 2. Update Records (Career or League)
            updateRecords(summary, winnerTeamId, winnerTeamName);

            // 3. Update League Table (if in league mode)
            if (gameState.currentMode === 'league') {
                const fixture = leagueState.fixtures.find(f => f.status === 'live');
                if (fixture) {
                    fixture.status = 'completed';
                    fixture.result = resultLine;
                    fixture.winnerKey = winnerTeamId;

                    // Find the preset keys for the fixture teams (using team names as identifier in leagueState)
                    const teamAKey = leagueState.teams.find(key => PRESET_TEAMS_DATA[key].name === gameState.teams['A'].name);
                    const teamBKey = leagueState.teams.find(key => PRESET_TEAMS_DATA[key].name === gameState.teams['B'].name);

                    // Pass the full match data needed for NRR calculation
                    const matchDataForNRR = { 
                        teams: JSON.parse(JSON.stringify(gameState.teams)),
                        gameMode: gameState.gameMode,
                        oversPerInnings: gameState.oversPerInnings,
                        target: gameState.target,
                        summary: gameState.summary
                    };
                    updateLeagueTable(winnerTeamId, resultLine, teamAKey, teamBKey, matchDataForNRR);
                }
            }

            // 4. Navigate to Summary Screen
            navigateTo('summary');
        };
        
        // --- LEAGUE LOGIC ---
        
        const startNewLeague = () => {
            // Check if league is already active but finished. If so, prompt to reset.
            const matchesRemaining = leagueState.fixtures.filter(f => f.status === 'upcoming').length;
            if (leagueState.isActive && matchesRemaining > 0) {
                // Resume logic handled by renderLeagueHome navigation
            } else {
                leagueState = getInitialLeagueState(); 
                leagueState.isActive = true;
                leagueState.leagueFormat = gameState.gameMode;
                leagueState.oversPerMatch = gameState.oversPerInnings;
            }

            saveState();
            navigateTo('league_home');
        };

        /** Calculates match run/ball data for NRR update. */
        const calculateMatchNRRData = (matchData, teamId) => {
            let runsF = 0;
            let ballsF = 0;
            let runsA = 0;
            let ballsA = 0;

            if (matchData.gameMode === 'test_match') {
                // NRR is typically zero in Test Matches (unless determined by rules we aren't implementing)
                return { RunsFor: 0, BallsFor: 0, RunsAgainst: 0, BallsAgainst: 0, isTest: true };
            }

            // Limited Overs NRR Logic
            const totalMaxBalls = matchData.oversPerInnings * 6;
            
            // Determine who batted first (Team A and B assignments are arbitrary but stable within the match)
            // Use InningsNum property to reliably determine 1st/2nd batting team regardless of A/B assignment
            const teamAInnings1Score = matchData.teams['A'].inningsScores.find(s => s.inningsNum === 1);
            const teamBInnings1Score = matchData.teams['B'].inningsScores.find(s => s.inningsNum === 1);
            
            let firstBattingTeamId, secondBattingTeamId;
            if (teamAInnings1Score) {
                firstBattingTeamId = 'A';
                secondBattingTeamId = 'B';
            } else { // Must be team B
                firstBattingTeamId = 'B';
                secondBattingTeamId = 'A';
            }
            
            const team1Score = matchData.teams[firstBattingTeamId].inningsScores.find(s => s.inningsNum === 1);
            const team2Score = matchData.teams[secondBattingTeamId].inningsScores.find(s => s.inningsNum === 2);
            
            if (!team1Score || !team2Score) {
                console.error("Incomplete scores for NRR calculation.");
                return { RunsFor: 0, BallsFor: 0, RunsAgainst: 0, BallsAgainst: 0, isTest: false };
            }

            const isCurrentTeamFirstBatting = teamId === firstBattingTeamId;
            const currentScore = isCurrentTeamFirstBatting ? team1Score : team2Score;
            const opponentScore = isCurrentTeamFirstBatting ? team2Score : team1Score;

            runsF = currentScore.runs;
            ballsF = oversToBalls(currentScore.overs, currentScore.balls);
            runsA = opponentScore.runs;
            ballsA = oversToBalls(opponentScore.overs, opponentScore.balls);

            // NRR Rule 1: If the chasing team wins, runs scored in the chase are only counted up to (Target - 1).
            if (!isCurrentTeamFirstBatting && currentScore.runs >= matchData.target) {
                runsF = matchData.target - 1; 
                // Balls faced remains actual balls faced.
            }
            
            // NRR Rule 2: If a team is All Out (wickets = 10), they are considered to have faced the full quota of overs, 
            // UNLESS the innings finished early due to target being reached.
            
            // Current Team (Runs For calculation)
            if (currentScore.wickets === 10 && currentScore.runs < matchData.target) {
                ballsF = totalMaxBalls;
            }
            
            // Opponent Team (Runs Against calculation)
            if (opponentScore.wickets === 10 && opponentScore.runs < (isCurrentTeamFirstBatting ? matchData.target : (runsF + 1))) {
                ballsA = totalMaxBalls;
            }

            return { RunsFor: runsF, BallsFor: ballsF, RunsAgainst: runsA, BallsAgainst: ballsA, isTest: false };
        };


        /** Updates the league table based on a completed match result, including NRR. */
        const updateLeagueTable = (winnerTeamId, result, teamAKey, teamBKey, matchDataForNRR) => {
            
            // Team A and Team B in fixture context (using keys like 'India', 'Australia')
            const teamAFixture = leagueState.table.find(t => t.teamKey === teamAKey);
            const teamBFixture = leagueState.table.find(t => t.teamKey === teamBKey);
            
            if (!teamAFixture || !teamBFixture) {
                 console.error("League team keys not found for table update.");
                 return;
            }
            
            // Calculate NRR components based on the match data in gameState
            const teamANRRData = calculateMatchNRRData(matchDataForNRR, 'A');
            const teamBNRRData = calculateMatchNRRData(matchDataForNRR, 'B');

            // Update stats for Team A (Fixture)
            teamAFixture.P += 1;
            teamAFixture.RunsFor += teamANRRData.RunsFor;
            teamAFixture.BallsFor += teamANRRData.BallsFor;
            teamAFixture.RunsAgainst += teamANRRData.RunsAgainst;
            teamAFixture.BallsAgainst += teamANRRData.BallsAgainst;

            // Update stats for Team B (Fixture)
            teamBFixture.P += 1;
            teamBFixture.RunsFor += teamBNRRData.RunsFor;
            teamBFixture.BallsFor += teamBNRRData.BallsFor;
            teamBFixture.RunsAgainst += teamBNRRData.RunsAgainst;
            teamBFixture.BallsAgainst += teamBNRRData.BallsAgainst;

            // Apply points and win/loss records
            const winnerName = winnerTeamId ? matchDataForNRR.teams[winnerTeamId].name : null;
            const matchFormat = matchDataForNRR.gameMode;

            if (result.includes("The match was tied") || (matchFormat === 'test_match' && result.includes("The match was drawn"))) {
                teamAFixture.T += 1;
                teamBFixture.T += 1;
                teamAFixture.Pts += (matchFormat === 'test_match' ? 1 : 1); // 1 point for T/D
                teamBFixture.Pts += (matchFormat === 'test_match' ? 1 : 1);
            } else if (winnerName === teamAFixture.name) {
                teamAFixture.W += 1;
                teamBFixture.L += 1;
                teamAFixture.Pts += (matchFormat === 'test_match' ? 4 : 2); // Test match wins get more points (4 standard)
            } else if (winnerName === teamBFixture.name) {
                teamBFixture.W += 1;
                teamAFixture.L += 1;
                teamBFixture.Pts += (matchFormat === 'test_match' ? 4 : 2);
            }

            // Calculate NRR for all teams in the table
            leagueState.table.forEach(t => {
                if (t.BallsFor > 0 && t.BallsAgainst > 0 && matchFormat !== 'test_match') {
                    const oversFor = ballsToOversDecimal(t.BallsFor);
                    const oversAgainst = ballsToOversDecimal(t.BallsAgainst);
                    
                    // Prevent division by zero
                    if (oversFor > 0 && oversAgainst > 0) {
                        const runRateFor = t.RunsFor / oversFor;
                        const runRateAgainst = t.RunsAgainst / oversAgainst;
                        t.NRR = (runRateFor - runRateAgainst).toFixed(3);
                    } else {
                        t.NRR = 0.000;
                    }
                } else {
                    t.NRR = 0.000; // Test matches and teams with no matches played
                }
            });

            // Sort table: Pts desc, then Wins desc, then NRR desc, then Name asc (Tiebreaker applied)
            leagueState.table.sort((a, b) => 
                b.Pts - a.Pts || 
                b.W - a.W || 
                parseFloat(b.NRR) - parseFloat(a.NRR) || 
                a.name.localeCompare(b.name)
            );
            
            saveState(); // Save league state immediately after updating table
        };
        
        /** Advances to the next match in the league after a match is completed. */
        const startNextLeagueMatch = () => {
            // 1. Clear volatile match state (scores, current innings, summary)
            clearMatchScoreboardState(); 
            
            // 2. Find and setup the NEXT UPCOMING fixture
            const nextIndex = leagueState.fixtures.findIndex(f => f.status === 'upcoming');
            
            if (nextIndex !== -1) {
                const nextFixture = leagueState.fixtures[nextIndex];
                setupLeagueMatch(nextFixture); // This handles all state setup and navigation to 'toss'
            } else {
                saveState();
                navigateTo('league_home');
            }
        };


        // --- SIMULATION CONTROL ---

        /** Simulates rolls until the end of the specified phase. */
        const simulateMatch = async (simulateUntil = 'match_end') => {
            if (gameState.isSimulating || gameState.matchResult) return;

            gameState.isSimulating = true;
            const initialInnings = gameState.innings; 
            saveState();
            renderScreen();

            const simulationLoop = async () => {
                // Stop conditions:
                if (!gameState.isSimulating || gameState.matchResult || gameState.currentScreen !== 'match') {
                    stopSimulation();
                    return;
                }
                
                let inningsEnded = await handleRoll();

                // If simulating until innings end, check if the innings number has advanced
                if (simulateUntil === 'innings_end' && inningsEnded && gameState.innings > initialInnings) {
                    stopSimulation();
                    // Ensure the innings break modal shows up properly
                    if (!gameState.matchResult) {
                        showInningsBreakMessage(); 
                    }
                    return;
                }
                
                if (gameState.matchResult) {
                    stopSimulation();
                    navigateTo('summary'); // Go straight to summary if match finished during simulation
                    return;
                }
                
                // If match active, render and continue
                if (gameState.isMatchActive) {
                    // Quick render update needed for score visualization
                    renderScreen();
                    setTimeout(simulationLoop, 10); // Very small delay for fast simulation
                }
            };
            
            simulationLoop();
        };
        
        const stopSimulation = () => {
            gameState.isSimulating = false;
            saveState();
            // If stopping mid-break, close modal if present, otherwise just re-render
            const modal = document.getElementById('innings-break-modal');
            if (modal) {
                // If modal is active, let the modal close function handle navigation/renderScreen
            } else {
                renderScreen(); 
            }
        };


        // --- SUMMARY & RECORDS LOGIC ---

        /** Calculates Man of the Tournament, Orange Cap, and Purple Cap winners. */
        const calculateLeagueAwards = (records) => {
            const playerStatsArray = Object.values(records.playerStats);

            let orangeCap = { name: 'N/A', runs: 0, teamName: '' };
            let purpleCap = { name: 'N/A', wickets: 0, runs: Infinity, teamName: '' };
            let manOfTheTournament = { name: 'N/A', score: -Infinity, reason: 'N/A', teamName: '' };
            let foundPlayer = false;

            playerStatsArray.forEach(p => {
                foundPlayer = true;
                // Orange Cap (Highest Runs)
                if (p.R > orangeCap.runs) {
                    orangeCap = { name: p.name, runs: p.R, teamName: p.teamName };
                }
                
                // Purple Cap (Highest Wickets, then lowest Runs conceded)
                // Need to calculate career Balls Bowled for proper Econ/rate comparison
                const totalOversDecimal = p.totalBallsBowled > 0 ? (Math.floor(p.totalBallsBowled / 6) + (p.totalBallsBowled % 6) / 6) : 0;
                const econ = totalOversDecimal > 0 ? p.Runs / totalOversDecimal : Infinity;

                if (p.W > purpleCap.wickets || (p.W === purpleCap.wickets && econ < purpleCap.runs)) {
                    purpleCap = { name: p.name, wickets: p.W, runs: p.Runs, teamName: p.teamName };
                }
                
                // Man of the Tournament (Impact Score: R * 0.7 + W * 10 - Runs Conceded * 0.3 + 100s * 50 + 50s * 20)
                const impact = (p.R * 0.7) + (p.W * 10) - (p.Runs * 0.3) + (p.hundreds * 50) + (p.fifties * 20);

                if (impact > manOfTheTournament.score) {
                    let reason = `Exceptional cumulative performance across the league, contributing significantly with both bat and ball.`;
                    
                    if (p.hundreds > 0 && p.W > 0) {
                         reason = `For a spectacular all-round performance, scoring ${p.R} runs (including ${p.hundreds} century(ies)) and taking ${p.W} wickets.`;
                    } else if (p.R > orangeCap.runs * 0.9 && p.R > 200) { // If near top scorer and played well
                         reason = `A dominating batting display throughout the league, finishing with ${p.R} runs.`;
                    } else if (p.W > purpleCap.wickets * 0.9 && p.W > 10) { // If near top wicket taker and bowled well
                         reason = `A consistent threat with the ball, finishing with ${p.W} wickets and a great economy rate.`;
                    } else if (p.R >= 150 || p.W >= 8) {
                        reason = `A monster tournament performance that dominated the headlines.`;
                    }

                    manOfTheTournament = {
                        name: p.name,
                        score: impact,
                        reason: reason,
                        teamName: p.teamName
                    };
                }
            });
            
            // Fallback for MOT if no stats accumulated
            if (!foundPlayer) {
                 manOfTheTournament = { name: 'N/A', score: 0, reason: 'No player data accumulated yet.', teamName: 'N/A' };
            }

            return { orangeCap, purpleCap, manOfTheTournament };
        };


        /** Generates the match summary data structure. */
        const generateMatchSummary = (winnerName, resultLine, firstBattingOverallTeamId, secondBattingOverallTeamId) => {
            const teamA = gameState.teams['A'];
            const teamB = gameState.teams['B'];

            const getInningsDetails = (teamId, inningsIndex) => { // inningsIndex 0 for 1st innings of that team, 1 for 2nd
                const team = gameState.teams[teamId];
                const inningsScore = team.inningsScores[inningsIndex]; // Specific innings score
                const inningsPlayersSnapshot = team.inningsScorecardHistory[inningsIndex]; // Specific innings player snapshot

                if (!inningsScore || !inningsPlayersSnapshot) return null;

                const topBatsmen = inningsPlayersSnapshot
                    .filter(p => p.R > 0)
                    .sort((a, b) => b.R - a.R)
                    .slice(0, 3)
                    .map(p => ({ name: p.name, runs: p.R, balls: p.B }));
                
                return {
                    name: team.name,
                    score: `${inningsScore.runs}/${inningsScore.wickets} (${inningsScore.overs}.${inningsScore.balls} Overs)`,
                    topBatsmen,
                };
            };

            let innings1Summary = null, innings2Summary = null, innings3Summary = null, innings4Summary = null;

            if (gameState.gameMode === 'limited_overs') {
                innings1Summary = getInningsDetails(firstBattingOverallTeamId, 0);
                innings2Summary = getInningsDetails(secondBattingOverallTeamId, 0);
            } else { // Test Match
                innings1Summary = getInningsDetails(gameState.testMatchState.team1Id, 0);
                innings2Summary = getInningsDetails(gameState.testMatchState.team2Id, 0);
                innings3Summary = getInningsDetails(gameState.testMatchState.team1Id, 1);
                innings4Summary = getInningsDetails(gameState.testMatchState.team2Id, 1);
            }

            const allPlayers = [...teamA.players, ...teamB.players];
            let bestImpact = { name: 'N/A', score: -Infinity, type: '', teamName: '', reason: 'No impactful statistics recorded.' };

            allPlayers.forEach(p => {
                if (p.B > 0 || p.Balls > 0) {
                    // Impact calculation: (Runs * 1.5) + (Wickets * 15) - (Runs Conceded * 0.5)
                    const impact = (p.R * 1.5) + (p.W * 15) - (p.Runs * 0.5);
                    const team = teamA.players.includes(p) ? teamA : teamB;

                    if (impact > bestImpact.score) {
                        
                        let reason = 'An exceptional all-round display.';
                        
                        // Check for dominance
                        if (p.R >= 50 && p.W >= 3) {
                             reason = `For an outstanding all-round display, scoring ${p.R} runs and taking ${p.W} wickets for ${p.Runs}.`;
                        } else if (p.R >= 50 && p.R * 1.5 > (p.W * 15)) {
                             reason = `For a crucial and match-winning innings of ${p.R} runs (${p.B} balls) including ${p.Fours} boundaries and ${p.Sixes} sixes.`;
                        } else if (p.W >= 3) {
                             reason = `For sensational bowling figures of ${p.W}/${p.Runs} in the match, restricting the opposition significantly.`;
                        } else if (p.R > 0 && p.W > 0) {
                             reason = `For an important contribution with both bat (${p.R} runs) and ball (${p.W}/${p.Runs}).`;
                        } else if (p.R > 0) {
                            reason = `For a crucial batting performance of ${p.R} runs.`;
                        } else if (p.W > 0) {
                            reason = `For their excellent work with the ball, taking ${p.W} wicket(s).`;
                        } else {
                            reason = 'Outstanding effort and sheer commitment throughout the match.';
                        }
                        
                        bestImpact = {
                            name: p.name,
                            score: impact,
                            type: (p.R > 0) ? (p.W > 0 ? 'All-Round' : 'Batting') : 'Bowling',
                            teamName: team.name,
                            reason: reason
                        };
                    }
                }
            });
            if (bestImpact.name === 'N/A' && allPlayers.length > 0) {
                const firstPlayerWhoPlayed = allPlayers.find(p => p.B > 0 || p.Balls > 0);
                if (firstPlayerWhoPlayed) {
                     const team = teamA.players.includes(firstPlayerWhoPlayed) ? teamA : teamB;
                     bestImpact = { name: firstPlayerWhoPlayed.name, score: 0, type: 'Participation', teamName: team.name, reason: 'N/A (Default MOM)' };
                } else {
                     bestImpact = { name: 'No Player', score: 0, type: 'N/A', teamName: 'N/A', reason: 'Match ended abruptly or no play occurred.' };
                }
            }

            // Calculate Top Bowlers for the entire match
            const allBowlersInMatch = allPlayers.filter(p => p.Balls > 0);
            const topMatchBowlers = allBowlersInMatch
                .sort((a, b) => b.W - a.W || a.Runs - b.Runs) // Sort by wickets, then runs conceded
                .slice(0, 5) // Get top 5 bowlers of the match
                .map(p => {
                    const teamName = teamA.players.includes(p) ? teamA.name : teamB.name;
                    return {
                        name: p.name,
                        teamName: teamName,
                        overs: `${Math.floor(p.Balls / 6)}.${p.Balls % 6}`,
                        wickets: p.W || 0,
                        runs: p.Runs || 0,
                        econ: (p.Balls > 0) ? (p.Runs / (p.Balls / 6)).toFixed(2) : '0.00'
                    };
                });
            
            return {
                matchDate: new Date().toLocaleDateString(),
                innings1: innings1Summary,
                innings2: innings2Summary,
                innings3: innings3Summary,
                innings4: innings4Summary,
                topMatchBowlers: topMatchBowlers, // Add the overall top bowlers to the summary
                resultLine: resultLine,
                manOfTheMatch: bestImpact,
                gameMode: gameState.gameMode,
                testLeadTrailMessage: gameState.gameMode === 'test_match' && !resultLine.includes("won by an innings") ? calculateTestLeadTrailMessage() : ''
            };
        };

        /** Updates the persistent career or league records. */
        const updateRecords = (summary, winnerTeamId, winnerTeamName) => {
            const isLeagueMatch = gameState.currentMode === 'league';
            const records = isLeagueMatch ? leagueState.records : careerRecords;
            const teamsToProcess = [gameState.teams['A'], gameState.teams['B']];

            // 1. Update Match History
            records.matchHistory.unshift({
                date: summary.matchDate,
                result: summary.resultLine,
                mom: summary.manOfTheMatch.name,
                gameMode: summary.gameMode,
                teamA: gameState.teams['A'].name,
                teamB: gameState.teams['B'].name
            });

            // 2. Calculate Fifties and Hundreds
            const matchFifties = new Map();
            const matchHundreds = new Map();

            teamsToProcess.forEach(team => {
                team.inningsScorecardHistory.forEach(inningsSnapshot => {
                    inningsSnapshot.forEach(playerInInnings => {
                        const playerKey = `${playerInInnings.name}_${team.name}`;
                        
                        // Check if player has accumulated stats in career/league records
                        if (records.playerStats[playerKey]) {
                            // Only count fifties/hundreds if they occurred in the specific match innings (R >= 50 or R >= 100)
                            if (playerInInnings.R >= 100 && playerInInnings.R < 200) { // Assume highest score is max 6*360 balls
                                matchHundreds.set(playerKey, (matchHundreds.get(playerKey) || 0) + 1); 
                            } else if (playerInInnings.R >= 50 && playerInInnings.R < 100) {
                                matchFifties.set(playerKey, (matchFifties.get(playerKey) || 0) + 1);
                            }
                        }
                    });
                });
            });

            // 3. Update Player Career Stats and Match Totals
            const allPlayersInMatch = [...gameState.teams['A'].players, ...gameState.teams['B'].players];
            allPlayersInMatch.forEach(p => {
                if (p.B > 0 || p.Balls > 0) {
                    const team = gameState.teams['A'].players.includes(p) ? gameState.teams['A'] : gameState.teams['B'];
                    const teamName = team.name;
                    const key = `${p.name}_${teamName}`;

                    if (!records.playerStats[key]) {
                        records.playerStats[key] = {
                            name: p.name, teamName: teamName, R: 0, B: 0, W: 0, Runs: 0, Matches: 0, totalBallsBowled: 0,
                            topScore: 0, bestBowlingFigure: { wickets: 0, runs: Infinity }, fifties: 0, hundreds: 0
                        };
                    }
                    const stats = records.playerStats[key];
                    stats.R += p.R;
                    stats.B += p.B;
                    stats.W += p.W;
                    stats.Runs += p.Runs;
                    stats.Matches += 1;
                    stats.totalBallsBowled += p.Balls; // Accumulate total balls bowled

                    // Find the highest score across all innings played in this match for this player
                    const highestInningsScore = team.inningsScorecardHistory
                                                .flatMap(snapshot => snapshot.filter(sp => sp.name === p.name))
                                                .reduce((max, sp) => Math.max(max, sp.R), 0);
                    
                    if (highestInningsScore > stats.topScore) {
                         stats.topScore = highestInningsScore;
                    }

                    if (p.W > stats.bestBowlingFigure.wickets ||
                        (p.W === stats.bestBowlingFigure.wickets && p.Runs < stats.bestBowlingFigure.runs)) {
                        stats.bestBowlingFigure = { wickets: p.W, runs: p.Runs };
                    }
                    
                    // Add accumulated fifties/hundreds from the match

            const teamName = gameState.teams[teamId].name;

            if (newNameTrimmed === '' || newNameTrimmed === oldName) {

            }

                    if (records.playerStats[newKey]) {

                        if (oldStats.topScore > existingStats.topScore) {

        // Simple fetch retry mechanism with exponential backoff
        async function fetchWithRetry(url, options, retries = 3) {
            for (let i = 0; i < retries; i++) {
                try {
                    const response = await fetch(url, options);
                    if (!response.ok) {
                        let errorMessage = `HTTP error! status: ${response.status} - ${response.statusText}`;
                        try {
                            const errorBody = await response.json();
                            if (errorBody.error && errorBody.error.message) {
                                errorMessage += `: ${errorBody.error.message}`;
                            }
                        } catch (parseError) {}
                        throw new Error(errorMessage);
                    }
                    return response;
                } catch (error) {
                    if (i < retries - 1) {
                        const delay = Math.pow(2, i) * 1000 + Math.random() * 500;
                        console.warn(`Retrying fetch (attempt ${i + 1}/${retries})...`, error);
                        await new Promise(resolve => setTimeout(resolve, delay));
                    } else {
                        throw error;
                    }
                }
            }
        }
